<script>
  /* ----------------- 1. DATABASE SETUP ----------------- */
  let db;
  const request = indexedDB.open("NeetPgAnkiDB", 1);
  request.onupgradeneeded = (e) => {
    db = e.target.result;
    if(!db.objectStoreNames.contains("cards")) {
      db.createObjectStore("cards", { keyPath: "id", autoIncrement: true });
    }
  };
  request.onsuccess = (e) => {
    db = e.target.result;
    fetchDueDeck();
  };

  /* ----------------- 2. IMAGE OCCLUSION DRAWING ----------------- */
  const canvas = document.getElementById('editCanvas');
  const ctx = canvas.getContext('2d');
  let loadedImg = null;
  let masks = []; // [{x, y, w, h}] normalized 0..1
  let isDrawing = false;
  let startX = 0, startY = 0;

  document.getElementById('imageChooser').addEventListener('change', function(e) {
    const file = e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(evt) {
      loadedImg = new Image();
      loadedImg.onload = function() {
        document.getElementById('editorWrap').style.display = 'block';
        resizeAndDrawCanvas();
      };
      loadedImg.src = evt.target.result;
    };
    reader.readAsDataURL(file);
  });

  function resizeAndDrawCanvas() {
    if (!loadedImg) return;
    const maxW = Math.min(window.innerWidth - 64, 480);
    const scale = maxW / loadedImg.width;
    canvas.width = maxW;
    canvas.height = loadedImg.height * scale;
    drawEditor();
  }

  function drawEditor() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    if (loadedImg) {
      ctx.drawImage(loadedImg, 0, 0, canvas.width, canvas.height);
    }
    // Draw masks
    ctx.fillStyle = "rgba(225, 29, 72, 0.9)";
    ctx.strokeStyle = "#fda4af";
    ctx.lineWidth = 2;
    masks.forEach(m => {
      const rx = m.x * canvas.width;
      const ry = m.y * canvas.height;
      const rw = m.w * canvas.width;
      const rh = m.h * canvas.height;
      ctx.fillRect(rx, ry, rw, rh);
      ctx.strokeRect(rx, ry, rw, rh);
    });
  }

  // Pointer events support iPhone touch + Desktop mouse
  function getPos(e) {
    const rect = canvas.getBoundingClientRect();
    const clientX = e.touches ? e.touches[0].clientX : e.clientX;
    const clientY = e.touches ? e.touches[0].clientY : e.clientY;
    return {
      x: (clientX - rect.left) / canvas.width,
      y: (clientY - rect.top) / canvas.height
    };
  }

  canvas.addEventListener('pointerdown', (e) => {
    isDrawing = true;
    const pos = getPos(e);
    startX = pos.x;
    startY = pos.y;
  });

  canvas.addEventListener('pointermove', (e) => {
    if (!isDrawing) return;
    const pos = getPos(e);
    drawEditor();
    ctx.fillStyle = "rgba(225, 29, 72, 0.5)";
    const x = Math.min(startX, pos.x) * canvas.width;
    const y = Math.min(startY, pos.y) * canvas.height;
    const w = Math.abs(pos.x - startX) * canvas.width;
    const h = Math.abs(pos.y - startY) * canvas.height;
    ctx.fillRect(x, y, w, h);
  });

  canvas.addEventListener('pointerup', (e) => {
    if (!isDrawing) return;
    isDrawing = false;
    const pos = getPos(e);
    const x = Math.min(startX, pos.x);
    const y = Math.min(startY, pos.y);
    const w = Math.abs(pos.x - startX);
    const h = Math.abs(pos.y - startY);
    if (w > 0.03 && h > 0.03) { // Ignore accidental taps
      masks.push({ x, y, w, h });
    }
    drawEditor();
  });

  function clearLastMask() {
    masks.pop();
    drawEditor();
  }

  function clearAllMasks() {
    masks = [];
    drawEditor();
  }

  /* ----------------- 3. ANKI SM-2 SRS ALGORITHM ----------------- */
  function calculateSM2(card, grade) {
    // grade: 1: Again, 2: Hard, 3: Good, 4: Easy
    let { repetitions = 0, interval = 0, easeFactor = 2.5 } = card;

    if (grade >= 3) {
      if (repetitions === 0) {
        interval = 1; // 1 day
      } else if (repetitions === 1) {
        interval = 3; // 3 days
      } else {
        interval = Math.round(interval * easeFactor);
      }
      repetitions++;
    } else if (grade === 2) { // Hard
      interval = Math.max(1, Math.round(interval * 1.2));
    } else { // Again
      repetitions = 0;
      interval = 0; // due immediately / today
    }

    // Ease factor adjustment formula
    easeFactor = easeFactor + (0.1 - (5 - grade) * (0.08 + (5 - grade) * 0.02));
    if (easeFactor < 1.3) easeFactor = 1.3;

    const nextDueDate = new Date();
    nextDueDate.setHours(0,0,0,0);
    nextDueDate.setDate(nextDueDate.getDate() + interval);

    return {
      repetitions,
      interval,
      easeFactor,
      nextDueDate: nextDueDate.getTime()
    };
  }

  /* ----------------- 4. CARD MANAGEMENT ----------------- */
  function saveCardRecord() {
    const subject = document.getElementById('cardSubject').value.trim() || 'General';
    const prompt = document.getElementById('cardPrompt').value.trim();
    const explanation = document.getElementById('cardExplanation').value.trim();

    if (!prompt && !loadedImg) {
      alert("Please provide at least a question or an image!");
      return;
    }

    const cardData = {
      subject,
      prompt,
      explanation,
      image: loadedImg ? loadedImg.src : null,
      masks: [...masks],
      repetitions: 0,
      interval: 0,
      easeFactor: 2.5,
      nextDueDate: Date.now() // Due right now
    };

    const tx = db.transaction("cards", "readwrite");
    tx.objectStore("cards").add(cardData);
    tx.oncomplete = () => {
      alert("Card added to your SRS deck!");
      // Reset form
      document.getElementById('cardSubject').value = '';
      document.getElementById('cardPrompt').value = '';
      document.getElementById('cardExplanation').value = '';
      document.getElementById('imageChooser').value = '';
      document.getElementById('editorWrap').style.display = 'none';
      loadedImg = null;
      masks = [];
      setTab('study');
    };
  }

  /* ----------------- 5. STUDY LOGIC ----------------- */
  let dueQueue = [];
  let currentCard = null;

  function fetchDueDeck() {
    const tx = db.transaction("cards", "readonly");
    const store = tx.objectStore("cards");
    const req = store.getAll();
    req.onsuccess = () => {
      const allCards = req.result;
      const now = Date.now();
      // Filter due cards
      dueQueue = allCards.filter(c => !c.nextDueDate || c.nextDueDate <= now);
      document.getElementById('deckDueCount').innerText = `Due: ${dueQueue.length}`;
      loadNextCard();
    };
  }

  function loadNextCard() {
    if (dueQueue.length === 0) {
      document.getElementById('dispTag').innerText = "All Caught Up!";
      document.getElementById('dispFront').innerText = "🎉 Shaandar! Aaj ke saare cards complete ho gaye hain.";
      document.getElementById('studyImgWrap').style.display = 'none';
      document.getElementById('dispAnswerBox').style.display = 'none';
      document.getElementById('btnReveal').style.display = 'none';
      document.getElementById('srsControls').style.display = 'none';
      return;
    }

    currentCard = dueQueue[0];
    document.getElementById('dispTag').innerText = currentCard.subject;
    document.getElementById('dispFront').innerText = currentCard.prompt || 'Identify the obscured section:';
    document.getElementById('dispBack').innerText = currentCard.explanation || 'No additional note.';
    document.getElementById('dispAnswerBox').style.display = 'none';
    document.getElementById('btnReveal').style.display = 'block';
    document.getElementById('srsControls').style.display = 'none';

    // Render image & occlusion masks
    const imgWrap = document.getElementById('studyImgWrap');
    const masksLayer = document.getElementById('masksLayer');
    masksLayer.innerHTML = '';

    if (currentCard.image) {
      imgWrap.style.display = 'inline-block';
      const img = document.getElementById('studyImg');
      img.src = currentCard.image;

      if (currentCard.masks && currentCard.masks.length > 0) {
        currentCard.masks.forEach(m => {
          const div = document.createElement('div');
          div.className = 'occlusion-mask';
          div.style.left = (m.x * 100) + '%';
          div.style.top = (m.y * 100) + '%';
          div.style.width = (m.w * 100) + '%';
          div.style.height = (m.h * 100) + '%';
          // Tap mask to peek / reveal individual label
          div.onclick = () => div.classList.toggle('revealed');
          masksLayer.appendChild(div);
        });
      }
    } else {
      imgWrap.style.display = 'none';
    }
  }

  function revealCard() {
    document.getElementById('dispAnswerBox').style.display = 'block';
    document.getElementById('btnReveal').style.display = 'none';
    document.getElementById('srsControls').style.display = 'grid';

    // Uncover all masks
    document.querySelectorAll('.occlusion-mask').forEach(m => m.classList.add('revealed'));
  }

  function rateCard(grade) {
    if (!currentCard) return;
    const updatedSRS = calculateSM2(currentCard, grade);
    Object.assign(currentCard, updatedSRS);

    const tx = db.transaction("cards", "readwrite");
    tx.objectStore("cards").put(currentCard);
    tx.oncomplete = () => {
      if (grade === 1) {
        // Move to end of current queue
        const c = dueQueue.shift();
        dueQueue.push(c);
      } else {
        dueQueue.shift();
      }
      document.getElementById('deckDueCount').innerText = `Due: ${dueQueue.length}`;
      loadNextCard();
    };
  }

  /* ----------------- 6. BACKUP & RESTORE ----------------- */
  function exportBackup() {
    const tx = db.transaction("cards", "readonly");
    tx.objectStore("cards").getAll().onsuccess = (e) => {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(e.target.result));
      const dlAnchor = document.createElement('a');
      dlAnchor.setAttribute("href", dataStr);
      dlAnchor.setAttribute("download", `NEET_PG_Deck_Backup_${new Date().toISOString().slice(0,10)}.json`);
      document.body.appendChild(dlAnchor);
      dlAnchor.click();
      dlAnchor.remove();
    };
  }

  function importBackup() {
    const file = document.getElementById('importInput').files[0];
    if (!file) {
      alert("Pehle .json backup file select karein!");
      return;
    }
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const imported = JSON.parse(e.target.result);
        if (!Array.isArray(imported)) throw new Error("Invalid format");
        
        const tx = db.transaction("cards", "readwrite");
        const store = tx.objectStore("cards");
        imported.forEach(card => {
          delete card.id; // allow fresh auto-increment
          store.add(card);
        });
        tx.oncomplete = () => {
          alert(`Successfully imported ${imported.length} cards!`);
          setTab('study');
        };
      } catch(err) {
        alert("Backup file corrupt ya invalid hai!");
      }
    };
    reader.readAsText(file);
  }

  /* ----------------- 7. UI TABS ----------------- */
  function setTab(tab) {
    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
    document.querySelectorAll('.card-panel').forEach(p => p.classList.remove('active'));
    if (tab === 'study') {
      document.querySelectorAll('.tab-btn')[0].classList.add('active');
      document.getElementById('studyTab').classList.add('active');
      fetchDueDeck();
    } else if (tab === 'add') {
      document.querySelectorAll('.tab-btn')[1].classList.add('active');
      document.getElementById('addTab').classList.add('active');
    } else {
      document.querySelectorAll('.tab-btn')[2].classList.add('active');
      document.getElementById('backupTab').classList.add('active');
    }
  }
</script>
